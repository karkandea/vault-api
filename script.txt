  let lastResults = [];

  // =========================================================
  // SESSION TRACKING
  // Menyimpan tanggal yang sukses selama tab browser masih ada.
  // =========================================================

  function loadSubmittedDates() {
    try {
      const stored = JSON.parse(
        sessionStorage.getItem(CONFIG.sessionKey) || "[]"
      );

      return new Set(Array.isArray(stored) ? stored : []);
    } catch {
      return new Set();
    }
  }

  const submittedDates = loadSubmittedDates();

  function saveSubmittedDates() {
    sessionStorage.setItem(
      CONFIG.sessionKey,
      JSON.stringify([...submittedDates])
    );
  }

  // =========================================================
  // DATE HELPERS
  // =========================================================

  function toPortalIsoDate(dateString) {
    /*
     * Portal menerima UTC.
     * 2026-07-02 pukul 00.00 WIB menjadi:
     * 2026-07-01T17:00:00.000Z
     */
    return new Date(`${dateString}T00:00:00+07:00`).toISOString();
  }

  function getWeekday(dateString) {
    const [year, month, day] = dateString.split("-").map(Number);
    return new Date(Date.UTC(year, month - 1, day)).getUTCDay();
  }

  function formatDate(dateString) {
    return new Intl.DateTimeFormat("id-ID", {
      weekday: "long",
      day: "2-digit",
      month: "long",
      year: "numeric",
      timeZone: "Asia/Jakarta"
    }).format(new Date(`${dateString}T00:00:00+07:00`));
  }

  function sleep(milliseconds) {
    return new Promise(resolve => setTimeout(resolve, milliseconds));
  }

  // =========================================================
  // PAYLOAD
  // =========================================================

  function buildPayload(entry) {
    return {
      employeeId: CONFIG.employeeId,
      roleId: CONFIG.roleId,
      consentrationId: CONFIG.consentrationId,
      date: toPortalIsoDate(entry.date),
      userId: CONFIG.userId,

      dailyBacklogs: [
        {

          systemId: CONFIG.systemId,

          demandId: CONFIG.demandId,
          branchName: CONFIG.branchName,
          workingHours: CONFIG.workingHours,
          categoryId: CONFIG.categoryId,
          activityId: CONFIG.activityId,
          remarks: entry.remarks
        }
      ]
    };
  }

  // =========================================================
  // VALIDATION
  // =========================================================

  function validateEntries() {
    const errors = [];
    const dates = new Set();

    for (const entry of ENTRIES) {
      if (!/^\d{4}-\d{2}-\d{2}$/.test(entry.date)) {
        errors.push(`${entry.date}: format tanggal tidak valid.`);
      }

      if (dates.has(entry.date)) {
        errors.push(`${entry.date}: tanggal duplicate di script.`);
      }

      dates.add(entry.date);

      const weekday = getWeekday(entry.date);

      if (weekday === 0 || weekday === 6) {
        errors.push(`${entry.date}: tanggal jatuh pada weekend.`);
      }

      if (!entry.remarks || !entry.remarks.trim()) {
        errors.push(`${entry.date}: remarks kosong.`);
      }
    }

    if (dates.has("2026-07-01")) {
      errors.push("1 Juli tidak boleh ada karena sudah diinput manual.");
    }

    if (ENTRIES.length !== 22) {
      errors.push(
        `Jumlah entry seharusnya 22, tetapi ditemukan ${ENTRIES.length}.`
      );
    }

    return errors;
  }

  // =========================================================
  // RESPONSE HELPER
  // =========================================================

  async function readResponse(response) {
    const text = await response.text();

    if (!text) {
      return null;
    }

    try {
      return JSON.parse(text);
    } catch {
      return text;
    }
  }

  // =========================================================
  // SUBMIT ONE ENTRY
  // =========================================================

  async function submitEntry(entry) {
    const payload = buildPayload(entry);

    console.log(`⏳ Submit ${entry.date} — ${formatDate(entry.date)}`);

    try {
      const response = await fetch(CONFIG.endpoint, {
        method: "POST",
        credentials: "include",
        headers: {
          "Content-Type": "application/json",
          Accept: "application/json, text/plain, */*"
        },
        body: JSON.stringify(payload)
      });

      const responseData = await readResponse(response);

      if (!response.ok) {
        console.error(
          `❌ ${entry.date} — HTTP ${response.status}`,
          responseData
        );

        return {
          date: entry.date,
          success: false,
          status: response.status,
          response: responseData
        };
      }

      submittedDates.add(entry.date);
      saveSubmittedDates();

      console.log(`✅ ${entry.date} berhasil`, responseData);

      return {
        date: entry.date,
        success: true,
        status: response.status,
        response: responseData
      };
    } catch (error) {
      console.error(`❌ ${entry.date} — request error`, error);

      return {
        date: entry.date,
        success: false,
        status: null,
        error: error.message
      };
    }
  }

  // =========================================================
  // BULK SUBMIT
  // Otomatis skip tanggal yang sukses dari test/session.
  // =========================================================

  async function bulkSubmit() {
    const validationErrors = validateEntries();

    if (validationErrors.length) {
      console.error("Validasi gagal:", validationErrors);
      return;
    }

    const pendingEntries = ENTRIES.filter(
      entry => !submittedDates.has(entry.date)
    );

    if (!pendingEntries.length) {
      console.log("✅ Tidak ada entry tersisa untuk disubmit.");
      return [];
    }

    console.table(
      pendingEntries.map(entry => ({
        tanggal: entry.date,
        hari: formatDate(entry.date),
        remarks: entry.remarks
      }))
    );

    const confirmed = window.confirm(
      `Submit ${pendingEntries.length} entry timesheet Juli 2026?\n\n` +
      "Pastikan test 2 Juli sudah terlihat benar di portal."
    );

    if (!confirmed) {
      console.log("Bulk submit dibatalkan.");
      return;
    }

    const results = [];

    for (let index = 0; index < pendingEntries.length; index++) {
      const entry = pendingEntries[index];

      console.log(
        `\n[${index + 1}/${pendingEntries.length}] Memproses ${entry.date}`
      );

      const result = await submitEntry(entry);
      results.push(result);

      if (index < pendingEntries.length - 1) {
        await sleep(CONFIG.delayMs);
      }
    }

    lastResults = results;

    const succeeded = results.filter(result => result.success);
    const failed = results.filter(result => !result.success);

    console.log("\n================ SUMMARY ================");
    console.log(`✅ Berhasil: ${succeeded.length}`);
    console.log(`❌ Gagal: ${failed.length}`);
    console.log("=========================================");

    console.table(
      results.map(result => ({
        tanggal: result.date,
        status: result.success ? "BERHASIL" : "GAGAL",
        httpStatus: result.status,
        response:
          typeof result.response === "string"
            ? result.response
            : JSON.stringify(result.response ?? result.error ?? "")
      }))
    );

    if (failed.length) {
      console.warn(
        "Ada entry gagal. Periksa detailnya, lalu jalankan retryFailed()."
      );
    }

    return results;
  }

  // =========================================================
  // RETRY HANYA ENTRY YANG GAGAL
  // =========================================================

  async function retryFailed() {
    const failedDates = lastResults
      .filter(result => !result.success)
      .map(result => result.date);

    if (!failedDates.length) {
      console.log("Tidak ada entry gagal dari proses terakhir.");
      return [];
    }

    const entriesToRetry = ENTRIES.filter(entry =>
      failedDates.includes(entry.date)
    );

    const confirmed = window.confirm(
      `Retry ${entriesToRetry.length} entry yang gagal?`
    );

    if (!confirmed) {
      console.log("Retry dibatalkan.");
      return;
    }

    const results = [];

    for (let index = 0; index < entriesToRetry.length; index++) {
      const result = await submitEntry(entriesToRetry[index]);
      results.push(result);

      if (index < entriesToRetry.length - 1) {
        await sleep(CONFIG.delayMs);
      }
    }

    lastResults = results;
    console.table(results);

    return results;
  }
